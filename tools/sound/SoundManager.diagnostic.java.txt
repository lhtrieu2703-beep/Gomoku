package game.view;

import javax.sound.sampled.*;
import java.io.ByteArrayInputStream;
import java.util.concurrent.*;
import java.util.function.BooleanSupplier;
import java.util.concurrent.atomic.AtomicLong;

/**
 * Trình quản lý âm thanh cho game Caro:
 * Tự động tổng hợp sóng âm thanh (Audio Synthesis) cho tiếng gõ cờ và tiếng chiến thắng
 * mà không cần file ngoài, đảm bảo 100% chạy mượt trên mọi máy tính.
 */
public class SoundManager {
    private static final boolean SOUND_DIAGNOSTICS = Boolean.getBoolean("gomoku.sound.diagnostics");
    private static final AtomicLong SOUND_SEQUENCE = new AtomicLong();
    private static final ThreadLocal<String> SOUND_EVENT = new ThreadLocal<>();
    private static final ExecutorService SOUND_DIAG_LOG = SOUND_DIAGNOSTICS
            ? Executors.newSingleThreadExecutor(task -> {
                Thread thread = new Thread(task, "gomoku-sound-diag");
                thread.setDaemon(true);
                return thread;
            }) : null;
    private static volatile boolean soundEnabled = true;
    private static final ThreadPoolExecutor PLAYBACK = new ThreadPoolExecutor(
            2, 2, 30, TimeUnit.SECONDS, new ArrayBlockingQueue<>(8), task -> {
                Thread thread = new Thread(task, "gomoku-audio");
                thread.setDaemon(true);
                return thread;
            }, new ThreadPoolExecutor.DiscardOldestPolicy());
    static { PLAYBACK.allowCoreThreadTimeOut(true); }

    public static void setSoundEnabled(boolean enabled) {
        soundEnabled = enabled;
        diagnostic("GLOBAL", "SET_ENABLED", null, Boolean.toString(enabled), null);
    }

    public static boolean isSoundEnabled() {
        return soundEnabled;
    }

    /**
     * Phát tiếng "cách" đanh của quân cờ gỗ khi đặt xuống bàn cờ.
     */
    public static void playPlacePieceSound() {
        playPlacePieceSound(() -> true);
    }

    public static void playPlacePieceSound(BooleanSupplier currentGame) {
        String event = "PLACE-" + SOUND_SEQUENCE.incrementAndGet();
        long requested = System.nanoTime();
        diagnostic(event, "REQUEST", null, caller(), null);
        if (!soundEnabled) { diagnostic(event, "SKIP_MUTED", null, "", null); return; }

        PLAYBACK.execute(() -> {
            SOUND_EVENT.set(event);
            diagnostic(event, "WORKER", null, "queuedMs=" + (System.nanoTime() - requested) / 1_000_000.0, null);
            if (!soundEnabled || !currentGame.getAsBoolean()) { diagnostic(event, "SKIP_STALE_OR_MUTED", null, "", null); return; }
            try {
                float sampleRate = 44100;
                int durationMs = 60;
                int numSamples = (int) (sampleRate * durationMs / 1000);
                byte[] buffer = new byte[numSamples];

                // Tổng hợp âm thanh gõ gỗ: Tần số khoảng 800Hz giảm dần về 300Hz với hàm tắt dần nhanh
                for (int i = 0; i < numSamples; i++) {
                    double time = i / (double) sampleRate;
                    double freq = 800.0 - 500.0 * (i / (double) numSamples);
                    double decay = Math.exp(-time * 50.0); // Tắt dần rất nhanh để tạo tiếng gõ gọn
                    double sinVal = Math.sin(2.0 * Math.PI * freq * time);
                    buffer[i] = (byte) (sinVal * decay * 110);
                }

                if (currentGame.getAsBoolean()) {
                    diagnostic(event, "GET_CLIP_BEGIN", null, "", null);
                    Clip clip = AudioSystem.getClip();
                    diagnostic(event, "GET_CLIP_END", clip, "", null);
                    playClip(clip, buffer, sampleRate, currentGame);
                }
            } catch (Exception ignored) {
                diagnostic(event, "EXCEPTION", null, "", ignored);
            }
        });
    }

    /**
     * Phát âm thanh chuông reo vui vẻ khi người chơi chiến thắng.
     */
    public static void playVictorySound() {
        playVictorySound(() -> true);
    }

    public static void playVictorySound(BooleanSupplier currentResult) {
        String event = "VICTORY-" + SOUND_SEQUENCE.incrementAndGet();
        long requested = System.nanoTime();
        diagnostic(event, "REQUEST", null, caller(), null);
        if (!soundEnabled) { diagnostic(event, "SKIP_MUTED", null, "", null); return; }

        PLAYBACK.execute(() -> {
            SOUND_EVENT.set(event);
            diagnostic(event, "WORKER", null, "queuedMs=" + (System.nanoTime() - requested) / 1_000_000.0, null);
            if (!soundEnabled || !currentResult.getAsBoolean()) { diagnostic(event, "SKIP_STALE_OR_MUTED", null, "", null); return; }
            try {
                float sampleRate = 44100;
                int durationMs = 400;
                int numSamples = (int) (sampleRate * durationMs / 1000);
                byte[] buffer = new byte[numSamples];

                // Hợp âm 3 nốt vui tươi (Do - Mi - Sol): 523Hz, 659Hz, 784Hz
                for (int i = 0; i < numSamples; i++) {
                    double time = i / (double) sampleRate;
                    double decay = Math.exp(-time * 4.0);
                    double sample = 0.4 * Math.sin(2.0 * Math.PI * 523.25 * time)
                                  + 0.3 * Math.sin(2.0 * Math.PI * 659.25 * time)
                                  + 0.3 * Math.sin(2.0 * Math.PI * 783.99 * time);
                    buffer[i] = (byte) (sample * decay * 110);
                }

                if (currentResult.getAsBoolean()) {
                    try {
                        diagnostic(event, "GET_CLIP_BEGIN", null, "", null);
                        Clip clip = AudioSystem.getClip();
                        diagnostic(event, "GET_CLIP_END", clip, "", null);
                        playClip(clip, buffer, sampleRate, currentResult);
                    }
                    catch (Exception ignored) { diagnostic(event, "EXCEPTION", null, "", ignored); }
                }
            } catch (Exception ignored) {
                diagnostic(event, "EXCEPTION", null, "", ignored);
            }
        });
    }

    /** Package-visible to verify cleanup without requiring an audio device. */
    static void playClip(Clip clip, byte[] buffer, float sampleRate) throws Exception {
        playClip(clip, buffer, sampleRate, () -> true);
    }

    static void playClip(Clip clip, byte[] buffer, float sampleRate, BooleanSupplier currentResult) throws Exception {
        String event = SOUND_EVENT.get() == null ? "DIRECT" : SOUND_EVENT.get();
        CountDownLatch stopped = new CountDownLatch(1);
        AudioFormat format = new AudioFormat(sampleRate, 8, 1, true, false);
        try (AudioInputStream stream = new AudioInputStream(
                new ByteArrayInputStream(buffer), format, buffer.length)) {
            clip.addLineListener(line -> {
                if (line.getType() == LineEvent.Type.STOP) stopped.countDown();
                diagnostic(event, "LINE_" + line.getType(), clip, "lineFrame=" + line.getFramePosition(), null);
            });
            diagnostic(event, "OPEN_BEGIN", clip, "format=" + format + " bytes=" + buffer.length, null);
            clip.open(stream);
            diagnostic(event, "OPEN_END", clip, "", null);
            if (!currentResult.getAsBoolean()) { diagnostic(event, "SKIP_STALE_AFTER_OPEN", clip, "", null); return; }
            clip.start();
            diagnostic(event, "START_RETURN", clip, "", null);
            long durationMs = (long) (1000.0 * buffer.length / sampleRate);
            long deadline = System.nanoTime() + (durationMs + 1000) * 1_000_000L;
            while (currentResult.getAsBoolean() && System.nanoTime() < deadline)
                if (stopped.await(10, TimeUnit.MILLISECONDS)) break;
            diagnostic(event, "WAIT_END", clip, "stopObserved=" + (stopped.getCount() == 0), null);
        } finally {
            diagnostic(event, "CLOSE_BEGIN", clip, "", null);
            clip.close();
            diagnostic(event, "CLOSE_END", clip, "", null);
        }
    }

    private static String caller() {
        if (!SOUND_DIAGNOSTICS) return "";
        return StackWalker.getInstance().walk(frames -> frames
                .filter(frame -> !frame.getClassName().equals(SoundManager.class.getName()))
                .limit(5).map(Object::toString).toList().toString());
    }

    private static void diagnostic(String event, String phase, Clip clip, String detail, Throwable error) {
        if (!SOUND_DIAGNOSTICS) return;
        String state = clip == null ? "clip=none" : "clip=" + System.identityHashCode(clip)
                + " isOpen=" + clip.isOpen() + " isRunning=" + clip.isRunning()
                + " framePosition=" + clip.getFramePosition() + " frameLength=" + clip.getFrameLength();
        String message = "[SOUND-DIAG] time=" + java.time.Instant.now() + " ns=" + System.nanoTime()
                + " event=" + event + " phase=" + phase + " thread=" + Thread.currentThread().getName()
                + " edt=" + javax.swing.SwingUtilities.isEventDispatchThread() + " enabled=" + soundEnabled
                + " active=" + PLAYBACK.getActiveCount() + " queued=" + PLAYBACK.getQueue().size()
                + " " + state + " " + detail;
        // stderr I/O is deliberately off the EDT, even for a REQUEST captured on the EDT.
        SOUND_DIAG_LOG.execute(() -> {
            System.err.println(message);
            if (error != null) error.printStackTrace(System.err);
        });
    }
}
